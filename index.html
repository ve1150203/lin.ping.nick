function renderResults(dataList) { const container = document.getElementById('resultContainer'); container.innerHTML = ''; if (!dataList || dataList.length === 0) { container.innerHTML = "

查無資料

"; return; } dataList.forEach(item => { let card = document.createElement('div'); card.className = 'expense-card'; // 兼容各種欄位名稱，確保金額不會變 $0 let storeName = item['消費項目/商家'] || item['消費項目'] || item['項目'] || '消費項目'; let amount = Number(item['總金額'] || item['金額'] || item['金額(NT$)'] || 0); let dateStr = item['日期'] || item['應繳日期'] || ''; let payment = item['支付方式'] || '現金'; let bank = item['刷卡銀行'] || payment; let detailsHtml = ''; if (item.showDetailBtn && item.details && item.details.length > 0) { let detailRows = ''; item.details.forEach(d => { let unitPrice = Number(d['單價']) || 0; let qty = Number(d['數量']) || 1; let subtotal = unitPrice * qty; // 精準計算小計 detailRows += ` ${d['品名'] || ''} $${unitPrice.toLocaleString()} ${qty} $${subtotal.toLocaleString()} `; }); detailsHtml = `
${detailRows}
品名 單價 數量 小計
`; } card.innerHTML = `
${storeName} $${amount.toLocaleString()}
📅 ${dateStr} ｜ 💳 ${payment} (${bank})
${detailsHtml} `; container.appendChild(card); }); }
